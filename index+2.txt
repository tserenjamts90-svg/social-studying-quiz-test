<script>
    // Энд та өөрийн хүссэн асуултуудаа нэмж, хасаж болно
    const allQuizData = [
        {
            category: "Социологи",
            question: "Нийгмийн харилцааны үндсэн хэлбэр аль нь вэ?",
            options: ["Ганцаарчилсан байдал", "Харилцан үйлдэл ба харилцаа", "Зөвхөн эдийн засгийн ашиг сонирхол", "Технологийн хэрэглээ"],
            correct: 1
        },
        {
            category: "Логик",
            question: "Бүх хүн мөнх бус. Сократ бол хүн. Иймд Сократ мөнх бус гэсэн дүгнэлт нь ямар ухагдахуун бэ?",
            options: ["Индукц", "Дедукц", "Харьцуулалт", "Аналоги"],
            correct: 1
        },
        {
            category: "Эдийн засаг",
            question: "Зах зээлийн эдийн засгийн үед үнийг үндсэндээ юу тодорхойlodog вэ?",
            options: ["Засгийн газрын тогтоол", "Эрэлт ба нийлүүлэлт", "Үйлдвэрлэгчдийн холбоо", "Банкны хүү"],
            correct: 1
        },
        {
            category: "Улс төр",
            question: "Монгол Улсын төрийн эрх барих дээд байгууллага аль нь вэ?",
            options: ["Засгийн газар", "Улсын Их Хурал", "Дээд шүүх", "Үндэсний аюулгүй байдлын зөвлөл"],
            correct: 1
        },
        {
            category: "Эрх зүй",
            question: "Монгол Улсын одоогийн Үндсэн хууль хэдэн онд батлагдсан бэ?",
            options: ["1924 он", "1940 он", "1960 он", "1992 он"],
            correct: 3
        },
        {
            category: "Олон улсын харилцаа",
            question: "НҮБ (Нэгдсэн Үндэстний Байгууллага) хэдэн онд байгуулагдсан бэ?",
            options: ["1919 он", "1945 он", "1950 он", "1990 он"],
            correct: 1
        }
    ];

    let selectedCategory = "Бүгд";
    let activeQuestions = [];
    let currentQuestionIndex = 0;
    let score = 0;
    let answered = false;
    let timer;
    let timeLeft = 30;

    function selectCategory(category, btn) {
        selectedCategory = category;
        document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('selected'));
        btn.classList.add('selected');
    }

    function startQuiz() {
        if (selectedCategory === "Бүгд") {
            activeQuestions = [...allQuizData];
        } else {
            activeQuestions = allQuizData.filter(q => q.category === selectedCategory);
        }

        if (activeQuestions.length === 0) {
            alert("Энэ ангилалд одоогоор асуулт байхгүй байна!");
            return;
        }

        document.getElementById('start-screen').style.display = 'none';
        document.getElementById('quiz-screen').style.display = 'block';
        currentQuestionIndex = 0;
        score = 0;
        loadQuestion();
    }

    function loadQuestion() {
        clearInterval(timer);
        answered = false;
        document.getElementById('next-btn').style.display = "none";
        
        const currentQ = activeQuestions[currentQuestionIndex];
        document.getElementById('question-counter').innerText = `Асуулт ${currentQuestionIndex + 1}/${activeQuestions.length} (${currentQ.category})`;
        document.getElementById('question-text').innerText = currentQ.question;

        const optionsContainer = document.getElementById('options-container');
        optionsContainer.innerHTML = "";

        currentQ.options.forEach((option, index) => {
            const btn = document.createElement("button");
            btn.classList.add("option-btn");
            btn.innerText = option;
            btn.onclick = () => selectOption(index);
            optionsContainer.appendChild(btn);
        });

        startTimer();
    }

    function startTimer() {
        timeLeft = 30;
        document.getElementById('timer-text').innerText = `Хугацаа: ${timeLeft}с`;
        document.getElementById('timer-progress').style.width = "100%";
        
        timer = setInterval(() => {
            timeLeft--;
            document.getElementById('timer-text').innerText = `Хугацаа: ${timeLeft}с`;
            document.getElementById('timer-progress').style.width = (timeLeft / 30 * 100) + "%";

            if (timeLeft <= 0) {
                clearInterval(timer);
                autoSelectWrong();
            }
        }, 1000);
    }

    function autoSelectWrong() {
        if (answered) return;
        answered = true;
        const currentQ = activeQuestions[currentQuestionIndex];
        const buttons = document.getElementById('options-container').getElementsByClassName('option-btn');
        buttons[currentQ.correct].classList.add('correct');
        document.getElementById('next-btn').style.display = "block";
    }

    function selectOption(selectedIndex) {
        if (answered) return;
        answered = true;
        clearInterval(timer);

        const currentQ = activeQuestions[currentQuestionIndex];
        const buttons = document.getElementById('options-container').getElementsByClassName('option-btn');

        if (selectedIndex === currentQ.correct) {
            buttons[selectedIndex].classList.add('correct');
            score++;
        } else {
            buttons[selectedIndex].classList.add('wrong');
            buttons[currentQ.correct].classList.add('correct');
        }

        document.getElementById('next-btn').style.display = "block";
    }

    function nextQuestion() {
        currentQuestionIndex++;
        if (currentQuestionIndex < activeQuestions.length) {
            loadQuestion();
        } else {
            showResult();
        }
    }

    function showResult() {
        clearInterval(timer);
        document.getElementById('quiz-screen').style.display = 'none';
        document.getElementById('result-screen').style.display = 'block';
        document.getElementById('result-score').innerText = `Таны авсан оноо: ${score} / ${activeQuestions.length}`;
    }
</script>
