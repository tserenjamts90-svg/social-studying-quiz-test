        function initQuizRunner(title, questions, minutes) {
            quizQuestions = questions;
            currentQIndex = 0;
            userAnswers = {};
            timeRemaining = minutes * 60;

            document.getElementById('quizTitle').innerText = title;
            document.getElementById('quizCategoryBadge').innerText = `${questions.length} Асуулт`;

            loadQuestion(0);
            renderMatrix();
            startTimer();
        }

        function loadQuestion(index) {
            currentQIndex = index;
            document.getElementById('currentQNum').innerText = index + 1;
            const q = quizQuestions[index];
            document.getElementById('questionText').innerText = q.question;

            const optsContainer = document.getElementById('optionsContainer');
            optsContainer.innerHTML = q.options.map((opt, i) => {
                const isSelected = userAnswers[index] === i;
                return `
                    <button onclick="selectOption(${i})" class="w-full p-4 rounded-xl text-left border transition flex items-center justify-between ${
                        isSelected 
                        ? 'border-sky-500 bg-sky-50 dark:bg-sky-950/40 text-sky-700 dark:text-sky-300 font-semibold' 
                        : 'border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700/50 text-slate-700 dark:text-slate-200'
                    }">
                        <span>${String.fromCharCode(65 + i)}. ${opt}</span>
                        ${isSelected ? '<i class="fa-solid fa-circle-check text-sky-500"></i>' : ''}
                    </button>
                `;
            }).join('');

            // Controls visibility
            document.getElementById('prevQBtn').disabled = index === 0;
            if (index === quizQuestions.length - 1) {
                document.getElementById('nextQBtn').classList.add('hidden');
                document.getElementById('finishQBtn').classList.remove('hidden');
            } else {
                document.getElementById('nextQBtn').classList.remove('hidden');
                document.getElementById('finishQBtn').classList.add('hidden');
            }
            renderMatrix();
        }

        function selectOption(optIdx) {
            userAnswers[currentQIndex] = optIdx;

        function nextQuestion() {
            if (currentQIndex < quizQuestions.length - 1) loadQuestion(currentQIndex + 1);

        function prevQuestion() {
            if (currentQIndex > 0) loadQuestion(currentQIndex - 1);

        function renderMatrix() {
            const matrix = document.getElementById('questionMatrix');
            if (!matrix) return;
            matrix.innerHTML = quizQuestions.map((_, i) => {
                const isAnswered = userAnswers.hasOwnProperty(i);
                const isCurrent = currentQIndex === i;
                let bgClass = 'bg-slate-100 dark:bg-slate-700 text-slate-500';
                if (isAnswered) bgClass = 'bg-sky-500 text-white font-bold';
                if (isCurrent) bgClass += ' ring-2 ring-offset-2 ring-sky-500';

                return `
                    <button onclick="loadQuestion(${i})" class="w-8 h-8 rounded-lg text-xs flex items-center justify-center transition ${bgClass}">
                        ${i + 1}
                    </button>
                `;
            }).join('');
        }

        function startTimer() {
            clearInterval(quizTimerInterval);
            updateTimerDisplay();
            quizTimerInterval = setInterval(() => {
                timeRemaining--;
                updateTimerDisplay();
                if (timeRemaining <= 0) {
                    clearInterval(quizTimerInterval);
                    finishQuiz();
                }
            }, 1000);
        }

        function updateTimerDisplay() {
            const m = Math.floor(timeRemaining / 60);
            const s = timeRemaining % 60;
            document.getElementById('timerDisplay').innerText = `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }

                clearInterval(quizTimerInterval);
                showView('homeView');
            }
        }

        function finishQuiz() {
            clearInterval(quizTimerInterval);
            let score = 0;
            quizQuestions.forEach((q, i) => {
                if (userAnswers[i] === q.correct) score++;
            });

            const percent = Math.round((score / quizQuestions.length) * 100);
            
            document.getElementById('resScore').innerText = `${score} / ${quizQuestions.length}`;
            document.getElementById('resPercent').innerText = `${percent}%`;
            document.getElementById('resRightWrong').innerText = `${score} / ${quizQuestions.length - score}`;
            document.getElementById('resTimeTaken').innerText = document.getElementById('timerDisplay').innerText;

            // Render breakdown
            const bd = document.getElementById('answersBreakdown');
            bd.innerHTML = quizQuestions.map((q, i) => {
                const uAns = userAnswers[i];
                const isCorrect = uAns === q.correct;
                return `
                    <div class="p-4 rounded-xl border ${isCorrect ? 'border-emerald-200 bg-emerald-50/30 dark:bg-emerald-950/20' : 'border-red-200 bg-red-50/30 dark:bg-red-950/20'} space-y-2">

        // Student Stats logic
        function saveStudentResult(score, total) {
            let stats = JSON.parse(localStorage.getItem('sq_student_stats')) || {};
            const key = currentUser.emailOrPhone;
            if (!stats[key]) {
                stats[key] = { name: currentUser.name, email: currentUser.emailOrPhone, phone: currentUser.phone || '-', count: 0, totalScore: 0, totalPossible: 0 };
            }
            stats[key].count += 1;
            stats[key].totalScore += score;
            stats[key].totalPossible += total;
            localStorage.setItem('sq_student_stats', JSON.stringify(stats));
            renderStudentStats();
        }

        function renderStudentStats() {
            const table = document.getElementById('studentReportTable');
            if (!table) return;
            const stats = JSON.parse(localStorage.getItem('sq_student_stats')) || {};
            const keys = Object.keys(stats);

            if (keys.length === 0) {
                table.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-xs text-slate-400">Одоогоор сорил өгсөн сурагч байхгүй байна.</td></tr>`;
                return;
            }

            table.innerHTML = keys.map(k => {
                const s = stats[k];
                const avgPct = Math.round((s.totalScore / s.totalPossible) * 100);
                return `
                    <tr class="hover:bg-slate-50 dark:hover:bg-slate-700/30">
                        <td class="p-4 font-semibold text-slate-800 dark:text-slate-200">${s.name}</td>
                        <td class="p-4 text-slate-500">${s.email}</td>
                        <td class="p-4 text-slate-500">${s.phone}</td>
                        <td class="p-4 font-bold text-sky-600">${s.count} удаа</td>
                        <td class="p-4 font-bold text-emerald-500">${avgPct}%</td>
                    </tr>
                `;
            }).join('');

            const navAuth = document.getElementById('authNavArea');

            }

        function renderAdminQuestionList() {
            const container = document.getElementById('adminQuestionList');
            if (!container) return;
            const all = getCombinedQuestions();
            
            container.innerHTML = all.map(q => `
                <div class="p-3 bg-slate-50 dark:bg-slate-900 rounded-xl flex justify-between items-center text-xs">
                    <div>
                        <span class="font-bold text-sky-600 uppercase mr-2">[${q.category}]</span>
                        <span class="text-slate-700 dark:text-slate-300">${q.question}</span>
                    </div>
                    <button onclick="deleteQuestion(${q.id})" class="text-red-500 hover:text-red-700 p-1">
                        <i class="fa-solid fa-trash-can"></i>
                    </button>
                </div>
            `).join('');
        }

                saved = saved.filter(q => q.id !== id);
                localStorage.setItem('sq_custom_questions', JSON.stringify(saved));
                
                // If base question, hide locally
                baseQuestions = baseQuestions.filter(q => q.id !== id);

                renderAdminQuestionList();
                showNotification('Асуултыг устгалаа.', 'info');
            }
        }

        // UI Toast Notifications
        function showNotification(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            if (!container) return;

            const toast = document.createElement('div');
            let bg = 'bg-slate-800 text-white';
            if (type === 'success') bg = 'bg-emerald-500 text-white';
            if (type === 'error') bg = 'bg-red-500 text-white';
            if (type === 'warning') bg = 'bg-amber-500 text-white';

            toast.className = `px-4 py-3 rounded-2xl shadow-xl font-semibold text-xs sm:text-sm flex items-center gap-2 ${bg} transition transform duration-300 translate-y-2 opacity-0`;
            toast.innerHTML = `<i class="fa-solid fa-circle-info"></i> ${message}`;

            container.appendChild(toast);
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
